# Core dependencies for the job_portal Django project.
# Installed with:  python -m venv venv && venv\Scripts\pip install -r requirements.txt
#
# Optional/heavy extras (news transcription, scraping agent, file watcher)
# live in requirements-optional.txt.

# --- Web framework ---
# Django 5.1.x is NOT compatible with Python 3.14. On 3.14 every admin
# changelist raises:
#   AttributeError: 'super' object has no attribute 'dicts'
# from django/template/context.py __copy__ (it calls copy(super())).
#
# Upgraded 5.2 -> 6.0 (6.0.9) to match what production actually runs.
# Pinned to the 6.0 line rather than 6.1 deliberately: 6.1 is a feature
# release and combining a major upgrade with a feature release makes
# regressions hard to attribute.
#
# Django 6 removed the no-argument form of format_html(); all such calls in
# this codebase now use mark_safe() for static markup and format_html(.., ..)
# for interpolated values.
Django>=6.0.9,<6.1
asgiref>=3.12,<4
sqlparse==0.5.3
tzdata==2025.1

# --- Database driver ---
# PyMySQL is registered as MySQLdb in job_portal/__init__.py.
# Use this instead of mysqlclient, which needs a C toolchain to build.
PyMySQL==1.2.3
# PostgreSQL driver (psycopg 3). The [binary] extra ships prebuilt wheels,
# so no libpq or C toolchain is needed on Windows or cPanel.
psycopg[binary]==3.3.6

# --- API ---
# djangorestframework is installed and configured in settings.py, but the
# existing endpoints (accounts/, billing/, tenants/) are plain Django
# function views returning JsonResponse -- they do not use it yet.
djangorestframework==3.18.3
# Required for the Telegram mini app when it is served from a different
# origin than the API. Deny-by-default: set CORS_ALLOWED_ORIGINS in the
# environment.
django-cors-headers==4.9.0

# --- Configuration ---
python-dotenv==1.0.1

# --- HTTP / scraping (scrape_tenders, post_website_update, agent) ---
requests==2.32.3

soupsieve==2.6
lxml>=5.3
certifi==2025.1.31

idna==3.10

# --- Form fields (CountryField in jobs management commands) ---
django-countries==7.6.1
